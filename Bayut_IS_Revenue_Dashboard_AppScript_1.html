⚙️  Google Apps Script Setup  ▼
1
Open your Google Sheet → click Extensions → Apps Script.
2
Delete any existing code, then paste the script from the code box on the right.
3
Click Save (💾), then Deploy → New Deployment.
4
Set type to Web app. Under "Who has access" choose Anyone. Click Deploy and authorise.
5
Copy the Web app URL (ends with /exec) and paste it below. Click Connect.
ℹ
After any code changes, always do Deploy → Manage Deployments → ✏️ Edit → New Version → Deploy to push updates.
⚠ Make sure the sheet tab is named exactly Reported Sales Data and the header row is row 1 (match column names in the script).
// ─── Bayut IS Revenue Dashboard — Apps Script ─────────────────────────────── // Paste into Extensions → Apps Script → Code.gs function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('Reported Sales Data'); if (!sheet) return json({ error: 'Sheet "Reported Sales Data" not found' }); const data = sheet.getDataRange().getValues(); const headers = data[0].map(h => String(h).trim()); const rows = []; for (let i = 1; i < data.length; i++) { const row = data[i]; // Skip completely blank rows if (row.every(c => c === '' || c === null || c === undefined)) continue; const obj = {}; headers.forEach((h, j) => { const v = row[j]; // Convert Date objects to ISO string, keep numbers as-is if (v instanceof Date) obj[h] = Utilities.formatDate(v, 'Asia/Riyadh', 'yyyy-MM-dd'); else obj[h] = v; }); rows.push(obj); } return json({ data: rows, generated: new Date().toISOString() }); } catch(err) { return json({ error: err.message }); } } function json(payload) { return ContentService .createTextOutput(JSON.stringify(payload)) .setMimeType(ContentService.MimeType.JSON); }
